function donut(parts,total){
  const r=52,C=2*Math.PI*r;let acc=0;
  const arcs=parts.map((p,i)=>{const len=total?p.n/total*C:0;const s=`<circle cx="70" cy="70" r="${r}" fill="none" stroke="var(--c${i+1})" stroke-width="18" stroke-dasharray="${len.toFixed(2)} ${(C-len).toFixed(2)}" stroke-dashoffset="${(-acc).toFixed(2)}"/>`;acc+=len;return s}).join('');
  return `<svg width="140" height="140" viewBox="0 0 140 140" role="img" aria-label="Distribuição por origem"><g transform="rotate(-90 70 70)"><circle cx="70" cy="70" r="${r}" fill="none" stroke="var(--line2)" stroke-width="18"/>${arcs}</g><text x="70" y="68" text-anchor="middle" style="fill:var(--ink);font-size:24px;font-weight:800;font-family:var(--font)">${total}</text><text x="70" y="86" text-anchor="middle" style="fill:var(--muted);font-size:11px;font-family:var(--font)">leads</text></svg>`;
}

function salvarConvite(){
  const g=id=>document.getElementById(id)?document.getElementById(id).value.trim():'';
  const nome=g('i-nome'),email=g('i-email'),papel=g('i-papel'),cid=state.role==='agencia'?(g('i-cid')||null):CID_CLIENTE;
  document.getElementById('ei-nome').textContent=nome?'':'Informe o nome.';
  const okMail=/^\S+@\S+\.\S+$/.test(email);document.getElementById('ei-email').textContent=okMail?'':'Informe um e-mail válido.';
  if(!nome||!okMail)return;
  if((papel==='Admin do cliente'||papel==='Comercial'||papel==='Visualizador')&&!cid){document.getElementById('ei-email').textContent='Esse papel precisa de um cliente.';return}
  USERS.push({id:'u'+Date.now(),nome,email,papel,cid,min:null,status:'Convite pendente'});
  logar(`convidou <b>${esc(nome)}</b> como ${papel}`,cid);toast('Convite enviado para '+email);
  fecharModal();render();
}

  logar(`moveu <b>${esc(l.nome)}</b> para ${st}`,l.cliente);toast('Movido para '+st);render();
